Chapter covered the basics of OpenCL, as well as the different OpenCL abstraction models, and
a basic OpenCL program example.

A platform is a vendor OpenCL representation
A device is a hardware component. You can have multiple devices per platform.
A context communicates between the host and the devices. You generally want one context per platform.


API REFERENCE
-------------

cl_int clGetPlatformIDs
(
    0,
    NULL,
    cl_uint* num_platforms
)
sets num_platforms to the number of discovered platforms
returns status code

cl_int clGetPlatformIDs
(
    cl_uint num_entries,
    cl_platform_id* platforms,
    0
)
sets platforms to the first num_entries platforms
returns status code


cl_int clGetPlatformInfo

    cl_platform_id platform,
    cl_platform_info param_name,
    0,
    NULL,
    size_t* param_value_size_ret

sets param_value_size_ret to the size needed to store param_name
this is unnecessary if the queried parameter is a fixed size (such as CL_PLATFORM_NUMERIC_VERSION)
returns status code

    cl_platform_id platform,
    cl_platform_info param_name,

    NULL

returns status code

cl_program clCreateProgramWithSource(
    cl_context context,
    cl_uint count, //Number of lines in the program
    const char** strings, //Lines of the program
    const size_t* lengths, //Line lengths of the program.
        // 0 in the array will use null-terminated for that line
        // NULL will use null-terminated for all lines
    cl_int* errcode_ret
)
returns a program object containing the source code


cl_int clBuildProgram(
    cl_program program,
    cl_uint num_devices,
    const cl_device_id* device_list,
    const char* options,
    void (CL_CALLBACK* pfn_notify)(cl_program program, void* user_data),
    void* user_data
)
Compiles the program for the devices in device_list
The callback function pfn_notify will be called whenever the program's state changes
returns a status code

cl_int clEnqueueNDRangeKernel(
    cl_command_queue command_queue, //Specifies the target device

    cl_uint work_dim, //number of dimensions for work items (limited to 1, 2, or 3)
    const size_t* global_work_offset, //offset so that work item IDs don't start at 0
    const size_t* local_work_size, //num of work items per dimension
    cl_uint num_events_in_wait_List, //number of items in the wait list
    const cl_event* event_wait_list, //prereq items before this event can run
    cl_event* event //current status of the kernel
)
returns a status code

cl_mem clCreateBuffer(
    cl_context context, //visible for all devices associated with the context
    cl_mem_flags flags, //read-only, write-only, or read-write, etc
    size_t size, //size of the buffer
    void* host_ptr, //optional based on the flag; allows allocating, copying, or directly using host memory
    cl_int* errcode_ret //error response
)
returns the requested buffer memory object
